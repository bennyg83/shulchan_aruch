שלח כו' אפי' לא אמר כו'. ע' תוס' דב"ק ק"ד א' ד"ה שליח לכאורה נראה כו' ומיהו כו' אבל בתוס' דב"מ צ"ח ב' ד"ה ואמר חזרו בהם ע"ש וכ"כ הנ"י שם והרא"ש בב"ק וכל הפוסקים ודבריהם דב"ק נדחין דהא אנן קי"ל כשמואל כמ"ש הרי"ף דסוגיא כוותיה כמ"ש ולשמואל מאי תקנתיה כו' וכמ"ש בש"ע בס"ג וס"ד וקי"ל נמי כר"י ור"א דשליח בעדים הוי שליח אלמא דשמואל מודה לו וא"כ מאי קאמר מאי תקנתיה ומוכח ג"כ מב"בפ"ז ב' וכמ"ש תוס' שם ד"ה רבה וי"ל דשאני כו' אלא דאח"כ דחקו עצמן לקיים סברתם וע' תוס' שם ד"ה אין אבל אם כו' א"כ אם כו' אלא שאח"כ כ' ולפי מה שפירש כו' וראיה דמהני אפי' בכתב ושלא בפניו ממה שפליגי בדיוקני וכפי פירושם שאין מעידין רק ע"י הדיוקין ואין מכיר כת"י אבל במכיר או עדים משלחין אלא שתוס' ד"ה הנ"ל במסקנא דחקו עצמן וחזרו ממ"ש בתחילה מכח קושיא הנ"ל אבל לפימ"ש בב"מ שם השמעתא עולה כהוגן:
והוא שיכיר כו'. ע' תוס' דב"ק שם אי כו' ושם בשליח בע"פ:
אפי' שהשליח כו'. דחרש ש"ו חד דינא אית להו ואימעטו מקרא דקטן וערש"י בגטין כ"ג א' ד"ה דלאו בני כו' ועכו"ם ע' בב"מ ע"א ב' ואית ליה דינא דקטן רק לענין זכיה ע"ש ובב"ב שם בשלמא באיסר כו' וה"ה להנך:
אפי' לא יחד כו'. כ"כ הרי"ף לגי' הטור וע' ב"ח ומתני' רי"א בהשואל שם ביד שליחך כו' וכן כו' ומשמע דלא יחד דאל"כ היינו בנו וגם פשטא דלישנא כן:
והוא כו'. ע' תוס' דב"ק שם ד"ה הכי כו' וזה הקו' קשה גם על פר"ח ל"ל למימר להכי טרחו כו' וצ"ל כתירוצם אלמא אע"ג שגילה דעתו שישלח ע"י כמ"ש אינש מהימנא הוא כו' מ"מ רצונו במקום בטוח:
ע"י מי כו'. ע' בפ"ק דגטין כגון שהוחזק כפרן אע"ג דהפקיד לכתב בידו וכ"ש בכה"ג:
אא"כ כו'. כמ"ש בב"ב שם בקטן כנ"ל לשדורי כו'.
